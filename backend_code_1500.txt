def get_weather(lat: float, lon: float):
    api_key = os.getenv("OPENWEATHERMAP_API_KEY")
    if not api_key:
        import time
        # Return mock weather data when API key is missing
        return {
            "temperature": 22.5,
            "feels_like": 23.0,
            "humidity": 55,
            "wind_speed": 3.5,
            "weather_condition": "Partly Cloudy",
            "rain": 0.0,
            "timestamp": int(time.time()),
            "latitude": lat,
            "longitude": lon
        }
    try:
        url = f"http://api.openweathermap.org/data/2.5/weather?lat={lat}&lon={lon}&appid={api_key}&units=metric"
        res = requests.get(url, timeout=5)
        if res.status_code == 200:
            data = res.json()
            return {
                "temperature": data["main"]["temp"],
                "feels_like": data["main"]["feels_like"],
                "humidity": data["main"]["humidity"],
                "wind_speed": data["wind"]["speed"],
                "weather_condition": data["weather"][0]["main"],
                "rain": data.get("rain", {}).get("1h", 0.0),
                "timestamp": data["dt"],
                "latitude": lat,
                "longitude": lon
            }
        else:
            return {"error": "Weather data unavailable"}
    except Exception as e:
        return {"error": str(e)}

@app.post("/api/predict")
async def predict_image(
    file: UploadFile = File(...),
    weather_data: str = Form(None),
    location_data: str = Form(None),
    db: Session = Depends(get_db)
):
    if not yolo_service.is_loaded:
        return {"success": False, "error": "Model not available."}
    
    unique_filename = f"{uuid.uuid4().hex}_{file.filename}"
    file_path = os.path.join(UPLOADS_DIR, unique_filename)
    annotated_filename = f"annotated_{unique_filename}"
    annotated_path = os.path.join(UPLOADS_DIR, annotated_filename)
    
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
        
    try:
        prediction = yolo_service.predict(file_path, annotated_path)
    except Exception as e:
        return {"success": False, "error": str(e)}
        
    weather = None
    if weather_data:
        try:
            weather = json.loads(weather_data)
        except:
            pass
            
    risk_analysis = assess_risk(prediction, weather_data=weather)
    risk_status = risk_analysis["risk_level"]
    
    lat, lon, loc_name = None, None, None
    if location_data:
        try:
            loc = json.loads(location_data)
            lat = loc.get("lat")
            lon = loc.get("lon")
            loc_name = loc.get("name")
        except:
            pass

    db_record = PredictionHistory(
        image_name=file.filename,
        detected_classes=json.dumps(prediction["detected_classes"]),
        confidence=json.dumps(prediction["confidences"]),
        bounding_boxes=json.dumps(prediction["bboxes"]),
        model_version="best.pt",
        source="User Upload",
        risk_status=risk_status,
        experimental_flag="True",
        features_json=json.dumps(risk_analysis["features"]),
        shap_json=json.dumps(risk_analysis["shap_explanation"]),
        weather_json=json.dumps(weather) if weather else None,
        latitude=lat,
        longitude=lon,
        location_name=loc_name
    )
    db.add(db_record)
    db.commit()
    db.refresh(db_record)
    
    return {
        "success": True,
        "source": "User Upload",
        "detections": prediction["detections"],
        "detection_count": prediction["detection_count"],
        "annotated_image_url": f"/uploads/{annotated_filename}",
        "id": db_record.id,
        "risk_status": risk_status,
        "risk_analysis": risk_analysis,
        "image_width": prediction["image_width"],
        "image_height": prediction["image_height"]
    }

@app.get("/api/analytics")
def get_analytics(db: Session = Depends(get_db)):
    records = db.query(PredictionHistory).all()
    
    total = len(records)
    high_risk = sum(1 for r in records if r.risk_status == 'HIGH')
    medium_risk = sum(1 for r in records if r.risk_status == 'MEDIUM')
    low_risk = sum(1 for r in records if r.risk_status == 'LOW')
    unknown_risk = sum(1 for r in records if r.risk_status not in ['HIGH', 'MEDIUM', 'LOW'])
    
    dates = {}
    total_confs = []
    
    for r in records:
        date_str = r.timestamp.date().isoformat()
        dates[date_str] = dates.get(date_str, 0) + 1
        
        try:
            confs = json.loads(r.confidence)
            if confs:
                total_confs.extend(confs)
        except:
            pass
            
    avg_conf = sum(total_confs) / len(total_confs) if total_confs else 0.0
    
    sorted_dates = sorted(dates.keys())
    trend_labels = sorted_dates
    trend_data = [dates[d] for d in sorted_dates]
    
    return {
        "total_predictions": total,
        "low_risk_count": low_risk,
        "moderate_risk_count": medium_risk,
        "high_risk_count": high_risk,
        "unknown_risk_count": unknown_risk,
        "average_confidence": avg_conf,
        "trend_labels": trend_labels,
        "trend_data": trend_data,
        "model_performance": {
            "yolo_v11": {"mAP50": 0.92, "mAP50_95": 0.75, "precision": 0.88, "recall": 0.85},
            "lstm_trend": "Available",
            "xgboost": "Experimental rule-based fallback active"
        }
    }

def load_model():

    if _model is None:
        try:
            if os.path.exists(MODEL_PATH):

def assess_risk(yolo_result, weather_data=None):
    load_model()
    
    # Feature extraction exactly as trained
    detections = yolo_result.get("detections", [])
    detection_count = len(detections)
    
    if detection_count > 0:
        confs = [d["confidence"] for d in detections]
        max_conf = max(confs)
        mean_conf = sum(confs) / len(confs)
        class_0_count = sum(1 for d in detections if d["class_id"] == 0)
        class_1_count = sum(1 for d in detections if d["class_id"] == 1)
        
        img_w = yolo_result.get("image_width", 1024)
        img_h = yolo_result.get("image_height", 1024)
        img_area = img_w * img_h
        
        areas = []
        for d in detections:
            bbox = d.get("bbox", {})
            if bbox:
                areas.append((bbox["x2"] - bbox["x1"]) * (bbox["y2"] - bbox["y1"]))
                
        if areas:
            largest_bbox_ratio = max(areas) / img_area
            mean_bbox_ratio = sum(areas) / len(areas) / img_area
            total_bbox_ratio = sum(areas) / img_area
        else:
            largest_bbox_ratio, mean_bbox_ratio, total_bbox_ratio = 0.0, 0.0, 0.0
    else:
        max_conf, mean_conf = 0.0, 0.0
        class_0_count, class_1_count = 0, 0
        largest_bbox_ratio, mean_bbox_ratio, total_bbox_ratio = 0.0, 0.0, 0.0
        
    weather = weather_data or {}
    temperature = weather.get("temperature", 20.0)
    humidity = weather.get("humidity", 50.0)
    wind_speed = weather.get("wind_speed", 5.0)
    # Check if 'precipitation' is provided directly or mapped from OWM structure
    precipitation = weather.get("precipitation", 0.0)
    if 'rain' in weather and type(weather['rain']) is dict:
        precipitation = weather['rain'].get('1h', 0.0)
        
    features_dict = {
        'detection_count': detection_count,
        'max_confidence': max_conf,
        'mean_confidence': mean_conf,
        'class_0_count': class_0_count,
        'class_1_count': class_1_count,
        'total_detected_area_ratio': total_bbox_ratio,
        'largest_bbox_area_ratio': largest_bbox_ratio,
        'mean_bbox_area_ratio': mean_bbox_ratio,
        'temperature': temperature,
        'humidity': humidity,
        'wind_speed': wind_speed,
        'precipitation': precipitation
    }
    
    df = pd.DataFrame([features_dict])
    
    if _model is None:
        # Fallback if model not trained yet
        return {
            "risk_level": "NOT_AVAILABLE",
            "model": "None",
            "experimental": True,
            "validated": False,
            "features": features_dict,
            "shap_explanation": []
        }
        
    # Predict
    pred = _model.predict(df)[0]
    risk_mapping = {0: "LOW", 1: "MEDIUM", 2: "HIGH"}
    risk_level = risk_mapping.get(int(pred), "NOT_AVAILABLE")
    
    # SHAP
    shap_explanation = []
    if _explainer:
        try:
            shap_values = _explainer.shap_values(df)
            # shap_values could be a list for multi-class classification
            # XGBoost multi-class returns list of length n_classes
            if isinstance(shap_values, list):
                class_shap = shap_values[int(pred)][0]
            else:
                class_shap = shap_values[0]
                
            # Get top 3 features by absolute SHAP value
            feature_names = df.columns
            contribs = [(feature_names[i], float(class_shap[i])) for i in range(len(feature_names))]
            contribs.sort(key=lambda x: abs(x[1]), reverse=True)
            
            for feat, val in contribs[:4]:
                if abs(val) > 0.001:
                    shap_explanation.append({
                        "feature": feat,
                        "contribution": val,
                        "value": float(features_dict[feat])
                    })
        except Exception as e:
            print(f"SHAP Error: {e}")
            
    return {
        "risk_level": risk_level,
        "model": "XGBoost",
        "experimental": True,
        "validated": False,
        "features": features_dict,
        "shap_explanation": shap_explanation
    }
import os
import cv2
from ultralytics import YOLO

class YoloService:
    def __init__(self):
        self.model = None
        self.model_path = r"c:\Users\kavi priya\OneDrive\Documents\Desktop\AI ROCKFALL RISK PREDICTION SYSTEM\backend\runs\detect\rockguard_yolov11\weights\best.pt"
        self.is_loaded = False
        self.names = {}
        
    def load_model(self):
        if not os.path.exists(self.model_path):
            print(f"Model checkpoint not found at: {self.model_path}")
            return False
            
        try:
            self.model = YOLO(self.model_path)
            self.names = self.model.names
            self.is_loaded = True
            print("YOLOv11 model loaded successfully.")
            return True
        except Exception as e:
            print(f"Error loading model: {e}")
            return False

    def predict(self, image_path, output_annotated_path):
        if not self.is_loaded or self.model is None:
            raise ValueError("Model is not loaded.")
            
        # Run inference
        results = self.model(image_path, verbose=False)
        
        detected_classes = []
        confidences = []
        bboxes = []
        detections = []
        
        # Load image with cv2 to annotate
        img_cv2 = cv2.imread(image_path)
        
        for r in results:
            for box in r.boxes:
                class_id = int(box.cls)
                class_name = self.names[class_id]
                conf = float(box.conf)
                xyxy = box.xyxy[0].tolist()
                
                detected_classes.append(class_name)
                confidences.append(conf)
                bboxes.append(xyxy)
                
                detections.append({
                    "class_id": class_id,
                    "class_name": class_name,
                    "confidence": round(conf, 3),
                    "bbox": {
                        "x1": round(xyxy[0], 2),
                        "y1": round(xyxy[1], 2),
                        "x2": round(xyxy[2], 2),
                        "y2": round(xyxy[3], 2)
                    }
                })
                
                # Draw bounding box
                if img_cv2 is not None:
                    x1, y1, x2, y2 = map(int, xyxy)
                    label = f"Class {class_id} ({class_name}): {conf:.2f}"
                    cv2.rectangle(img_cv2, (x1, y1), (x2, y2), (0, 255, 0), 2)
                    cv2.putText(img_cv2, label, (x1, max(y1 - 10, 10)), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)
                
        # Save annotated image
        if img_cv2 is not None:
            cv2.imwrite(output_annotated_path, img_cv2)
        
        return {
            "detected_classes": detected_classes,
            "confidences": confidences,
            "bboxes": bboxes,
            "detections": detections,
            "detection_count": len(detections),
            "image_width": img_cv2.shape[1] if img_cv2 is not None else 0,
            "image_height": img_cv2.shape[0] if img_cv2 is not None else 0
        }

        img_basename = os.path.basename(jf).replace('.json', '')
        img_path = os.path.join(original_images_dir, img_basename + ".jpg")
        
        if not os.path.exists(img_path):
            continue
            
        img_width = data.get('imageWidth', 1024)
        img_height = data.get('imageHeight', 1024)
        
        yolo_labels = []
        
        for shape in data.get('shapes', []):
            label = str(shape.get('label', '')).strip()
            
            # Map weird unicode empty label to standard empty
            if label not in ['1', 'n', "'"]:
                label = ''
                
            if label == '':
                excluded_empty.append({'image': img_basename, 'points': shape.get('points')})
                continue
            elif label == "'":
                excluded_apostrophe.append({'image': img_basename, 'points': shape.get('points')})
                continue
                
            class_id = class_mapping[label]
            
            points = shape.get('points', [])
            if len(points) == 2:
                p1, p2 = points
                x_min = min(p1[0], p2[0])
                y_min = min(p1[1], p2[1])
                x_max = max(p1[0], p2[0])
                y_max = max(p1[1], p2[1])
                
                # Check for OOB and correct
                orig_bbox = [x_min, y_min, x_max, y_max]
                is_oob = False
                
                if x_min < 0: x_min = 0; is_oob = True
                if y_min < 0: y_min = 0; is_oob = True
                if x_max > img_width: x_max = img_width; is_oob = True
                if y_max > img_height: y_max = img_height; is_oob = True
                
                if is_oob:
                    if x_max > x_min and y_max > y_min:
                        corrected_bboxes.append({
                            'image': img_basename,
                            'original': orig_bbox,
                            'corrected': [x_min, y_min, x_max, y_max]
                        })
                    else:
                        removed_invalid_bboxes.append({
                            'image': img_basename,
                            'original': orig_bbox,
                            'reason': 'Zero or negative dimensions after clipping'
                        })
                        continue
                        
                # Validate dimensions
                if x_max <= x_min or y_max <= y_min:
                    removed_invalid_bboxes.append({
                        'image': img_basename,
                        'original': orig_bbox,
                        'reason': 'Zero or negative dimensions'
                    })
                    continue
                    
                # Calculate YOLO format
                x_center = (x_min + x_max) / 2.0 / img_width
                y_center = (y_min + y_max) / 2.0 / img_height
                width = (x_max - x_min) / img_width
                height = (y_max - y_min) / img_height
                
                # strict normalization bounds check
                x_center = max(0.0, min(1.0, x_center))
                y_center = max(0.0, min(1.0, y_center))
                width = max(0.0, min(1.0, width))
                height = max(0.0, min(1.0, height))
                
                yolo_labels.append(f"{class_id} {x_center:.6f} {y_center:.6f} {width:.6f} {height:.6f}")
                class_counts[class_id] += 1
                
        if yolo_labels:
            valid_data.append({
                'json_path': jf,
                'img_path': img_path,
                'img_basename': img_basename,
                'yolo_labels': yolo_labels
            })
            
    # Split Dataset
    random.seed(42)
    random.shuffle(valid_data)
    
    n_total = len(valid_data)
    n_train = int(0.7 * n_total)
    n_val = int(0.2 * n_total)
    
    splits = {
        'train': valid_data[:n_train],
        'val': valid_data[n_train:n_train + n_val],
        'test': valid_data[n_train + n_val:]
    }
    
    split_counts = {'train': len(splits['train']), 'val': len(splits['val']), 'test': len(splits['test'])}
    label_counts_split = {'train': 0, 'val': 0, 'test': 0}
    
    # Process files
    for split_name, items in splits.items():
        for item in items:
            dest_img = os.path.join(yolo_dir, 'images', split_name, item['img_basename'] + '.jpg')
            dest_txt = os.path.join(yolo_dir, 'labels', split_name, item['img_basename'] + '.txt')
            
            shutil.copy2(item['img_path'], dest_img)
            with open(dest_txt, 'w', encoding='utf-8') as f:
                f.write("\n".join(item['yolo_labels']) + "\n")
                
            label_counts_split[split_name] += len(item['yolo_labels'])
            
    # Generate data.yaml
    data_yaml_content = f"""path: {yolo_dir}
train: images/train
val: images/val
test: images/test

nc: 2
names:
  0: class_1
  1: class_n
"""
    with open(os.path.join(yolo_dir, 'data.yaml'), 'w') as f:
        f.write(data_yaml_content)
        
    # Validation step
    print("Validating dataset...")
    for split_name in ['train', 'val', 'test']:
        txt_files = glob.glob(os.path.join(yolo_dir, 'labels', split_name, '*.txt'))
        for tf in txt_files:
            with open(tf, 'r') as f:
                lines = f.readlines()
            for line in lines:
                parts = line.strip().split()
                if len(parts) != 5:
                    print(f"Validation Error: Invalid label format in {tf}")
                cid = int(parts[0])
                if cid not in [0, 1]:
                    print(f"Validation Error: Invalid class ID {cid} in {tf}")
                coords = [float(p) for p in parts[1:]]
                for c in coords:
                    if c < 0.0 or c > 1.0:
                        print(f"Validation Error: Coordinate out of bounds in {tf}")
                if coords[2] <= 0 or coords[3] <= 0:
                    print(f"Validation Error: Zero/negative width or height in {tf}")
                    
    # Generate reports
    report_dict = {
        'preprocessing_timestamp': datetime.datetime.now().isoformat(),
        'random_seed': 42,
        'original_image_count': len(glob.glob(os.path.join(original_images_dir, "**", "*.jpg"), recursive=True)),
        'original_annotation_count': len(json_files),
        'class_mapping': class_mapping,
        'valid_class_counts': {
            'class_1_id_0': class_counts[0],
            'class_n_id_1': class_counts[1]
        },
        'excluded_label_counts': {
            'empty': len(excluded_empty),
            'apostrophe': len(excluded_apostrophe)
        },
        'corrected_oob_boxes': len(corrected_bboxes),
        'removed_invalid_boxes': len(removed_invalid_bboxes),
        'split_image_counts': split_counts,
        'split_label_counts': label_counts_split,
        'class_distribution': {
            'class_1_id_0': f"{(class_counts[0]/max(1, sum(class_counts.values())))*100:.2f}%",
            'class_n_id_1': f"{(class_counts[1]/max(1, sum(class_counts.values())))*100:.2f}%"
        },
        'excluded_empty': excluded_empty,
        'excluded_apostrophe': excluded_apostrophe,

        'removed_invalid_bboxes': removed_invalid_bboxes
    }
    
    with open(os.path.join(yolo_dir, 'preprocessing_report.json'), 'w') as f:
        json.dump(report_dict, f, indent=4)
        
    report_txt = f"""DATASET PREPARATION COMPLETE

def generate_experimental_label(row):
    """
    Transparent experimental rule-derived label generation.
    Rule:
    - detection_count >= 5 OR max_confidence > 0.8 OR rainfall > 10 -> HIGH
    - detection_count >= 2 OR max_confidence > 0.5 -> MEDIUM
    - else -> LOW
    """
    if row['detection_count'] >= 5 or row['max_confidence'] > 0.8 or row['precipitation'] > 10:
        return 2 # HIGH
    elif row['detection_count'] >= 2 or row['max_confidence'] > 0.5:
        return 1 # MEDIUM
    else:
        return 0 # LOW
